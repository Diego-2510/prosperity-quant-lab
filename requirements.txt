# Runtime dependencies for the Prosperity-4 backtesting engine.
# Tested on Python 3.10-3.12.

# --- Core (required) ---------------------------------------------------------
numpy>=1.24
pandas>=2.0
matplotlib>=3.7

# --- Optional but strongly recommended ---------------------------------------

# to single-threaded execution if missing, but installing them typically
# delivers a 6-14x speedup on the grid-search and Monte-Carlo eval stages.
joblib>=1.3
tqdm>=4.65

# scipy: only used in bs_implied_vol() to access scipy.stats.norm for the
# Black-Scholes vega Newton step. The code falls back to a hand-rolled erf
# implementation if scipy is unavailable, so this is technically optional --
# but the scipy path is roughly 5x faster on large IV-inversion runs.
scipy>=1.10
